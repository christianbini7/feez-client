-- ══════════════════════════════════════════════════
-- FEEZ — Setup initial
-- Exécuter dans : Supabase Dashboard → SQL Editor
-- ══════════════════════════════════════════════════

-- Extensions nécessaires
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";       -- UUIDs auto
CREATE EXTENSION IF NOT EXISTS "pgcrypto";        -- Chiffrement
CREATE EXTENSION IF NOT EXISTS "postgis";         -- GPS / géolocalisation
CREATE EXTENSION IF NOT EXISTS "pg_trgm";         -- Recherche texte rapide

-- Fonction utilitaire : timestamp automatique
CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Fonction : générer un numéro de commande lisible
CREATE OR REPLACE FUNCTION generate_order_ref()
RETURNS TEXT AS $$
BEGIN
  RETURN 'FZ-' || LPAD(FLOOR(RANDOM() * 9000 + 1000)::TEXT, 4, '0');
END;
$$ LANGUAGE plpgsql;

-- ══ TABLES ADMIN ROLES & PERMISSIONS ══

CREATE TABLE admin_roles (
  id           uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  name         text NOT NULL UNIQUE,   -- 'super_admin', 'ops', 'catalog'…
  display_name text NOT NULL,
  level        int2 NOT NULL,           -- 1=Super, 2=Admin, 3=Manager
  color        text,                     -- Couleur dans le dashboard
  created_at   timestamptz DEFAULT NOW()
);

CREATE TABLE admin_permissions (
  id           uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  role_id      uuid REFERENCES admin_roles(id) ON DELETE CASCADE,
  resource     text NOT NULL,           -- 'products', 'orders', 'riders'…
  can_view     bool DEFAULT false,
  can_create   bool DEFAULT false,
  can_edit     bool DEFAULT false,
  can_delete   bool DEFAULT false,
  UNIQUE(role_id, resource)
);

CREATE TABLE admin_users (
  id           uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  auth_id      uuid REFERENCES auth.users(id) ON DELETE CASCADE UNIQUE,
  role_id      uuid REFERENCES admin_roles(id),
  first_name   text NOT NULL,
  last_name    text NOT NULL,
  email        text NOT NULL UNIQUE,
  phone        text,
  avatar_url   text,
  is_active    bool DEFAULT true,
  last_login   timestamptz,
  created_at   timestamptz DEFAULT NOW(),
  updated_at   timestamptz DEFAULT NOW()
);

CREATE TABLE admin_audit_logs (
  id           uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  admin_id     uuid REFERENCES admin_users(id),
  action       text NOT NULL,           -- 'CREATE', 'UPDATE', 'DELETE'
  resource     text NOT NULL,           -- Table concernée
  resource_id  text,                     -- ID de l'enregistrement
  old_data     jsonb,                    -- Données avant modification
  new_data     jsonb,                    -- Données après modification
  ip_address   text,
  created_at   timestamptz DEFAULT NOW()
);

CREATE TABLE categories (
  id           uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  name         text NOT NULL UNIQUE,   -- 'epicerie', 'pharmacie'…
  display_name text NOT NULL,
  icon         text,                     -- emoji ou URL icône

CREATE TABLE products (

  category_id     uuid REFERENCES categories(id),
  name            text NOT NULL,
  description     text,
  unit            text,               -- 'Pièce', '500g', 'Boîte 30'…

  emoji           text,               -- Fallback si pas d'image

  created_by      uuid REFERENCES admin_users(id),
  created_at      timestamptz DEFAULT NOW(),

);

CREATE INDEX idx_products_avail ON products(is_available);

Stocks & inventaire

SQLinventory.sql
CREATE TABLE inventory (
  id              uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  product_id      uuid REFERENCES products(id) ON DELETE CASCADE UNIQUE,
  quantity        int4 NOT NULL DEFAULT 0,
  reserved        int4 DEFAULT 0,      -- Réservé par commandes en cours
  alert_threshold int4 DEFAULT 5,      -- Alerte si quantité < seuil
  last_restock    timestamptz,
  updated_at      timestamptz DEFAULT NOW()
);

-- Vue disponible réelle (quantité - réservé)
CREATE VIEW inventory_available AS
  SELECT
    i.*,
    p.name, p.emoji, p.price, p.category_id,
    (i.quantity - i.reserved) AS available,
    (i.quantity - i.reserved) < i.alert_threshold AS low_stock
  FROM inventory i
  JOIN products p ON i.product_id = p.id;

CREATE TABLE inventory_movements (
  id           uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  product_id   uuid REFERENCES products(id),
  type         text NOT NULL,         -- 'in', 'out', 'adjustment', 'reserved'
  quantity     int4 NOT NULL,
  reason       text,                   -- 'order_#FZ-4847', 'restock', 'loss'…
  admin_id     uuid REFERENCES admin_users(id),
  created_at   timestamptz DEFAULT NOW()
);

-- Fonction : mettre à jour le stock automatiquement après une commande
CREATE OR REPLACE FUNCTION decrement_stock(p_id uuid, qty int4)
RETURNS void AS $$
BEGIN
  UPDATE inventory
  SET quantity = quantity - qty, updated_at = NOW()
  WHERE product_id = p_id;
END;
$$ LANGUAGE plpgsql;

Commandes

SQLorders.sql
CREATE TABLE orders (
  id              uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  ref             text DEFAULT generate_order_ref() UNIQUE, -- FZ-4847
  type            text NOT NULL,       -- 'market' ou 'food'
  user_id         uuid REFERENCES users(id),
  rider_id        uuid REFERENCES riders(id),
  restaurant_id   uuid REFERENCES restaurants(id),

  -- Statut (pipeline complet)
  status          text NOT NULL DEFAULT 'pending',
  -- pending → accepted → preparing → ready → in_transit → delivered
  -- OU : pending → cancelled / rejected

  -- Articles (snapshot au moment de la commande)
  items           jsonb NOT NULL,      -- [{product_id, name, qty, price}]

  -- Montants
  subtotal        int4 NOT NULL,
  delivery_fee    int4 DEFAULT 200,
  discount        int4 DEFAULT 0,
  total           int4 NOT NULL,

  -- Livraison
  delivery_address text,
  delivery_lat    float8,
  delivery_lng    float8,
  delivery_zone   text,
  delivery_notes  text,

  -- Paiement
  payment_method  text,               -- 'wave', 'orange', 'mtn', 'card', 'cash'
  payment_status  text DEFAULT 'pending', -- pending / paid / failed / refunded
  payment_ref     text,               -- Référence CinetPay
  promo_code      text,

  -- Timestamps pipeline
  accepted_at     timestamptz,
  preparing_at    timestamptz,
  ready_at        timestamptz,
  transit_at      timestamptz,
  delivered_at    timestamptz,
  cancelled_at    timestamptz,
  cancel_reason   text,

  created_at      timestamptz DEFAULT NOW()
);

CREATE INDEX idx_orders_user   ON orders(user_id);
CREATE INDEX idx_orders_rider  ON orders(rider_id);
CREATE INDEX idx_orders_status ON orders(status);
CREATE INDEX idx_orders_date   ON orders(created_at DESC);

Row Level Security (RLS)

Les policies RLS garantissent que chaque utilisateur ne voit que ses propres données. C'est la sécurité côté base de données.

🔐

Principe RLS

Même si quelqu'un obtient ta clé Supabase, il ne peut pas lire les commandes des autres. RLS s'applique au niveau PostgreSQL — impossible à contourner depuis le client Flutter.

SQLrls_policies.sql
-- ══ Activer RLS sur toutes les tables ══
ALTER TABLE users                ENABLE ROW LEVEL SECURITY;
ALTER TABLE user_addresses      ENABLE ROW LEVEL SECURITY;
ALTER TABLE products            ENABLE ROW LEVEL SECURITY;
ALTER TABLE inventory           ENABLE ROW LEVEL SECURITY;
ALTER TABLE orders              ENABLE ROW LEVEL SECURITY;
ALTER TABLE restaurants         ENABLE ROW LEVEL SECURITY;
ALTER TABLE riders              ENABLE ROW LEVEL SECURITY;
ALTER TABLE payments            ENABLE ROW LEVEL SECURITY;

-- ══ USERS : chaque user voit seulement son profil ══
CREATE POLICY "users_own_data" ON users
  FOR ALL USING (auth.uid() = id);

-- ══ ADRESSES : chaque user voit ses adresses ══
CREATE POLICY "addresses_own" ON user_addresses
  FOR ALL USING (auth.uid() = user_id);

-- ══ PRODUITS : tout le monde peut lire les produits disponibles ══
CREATE POLICY "products_public_read" ON products
  FOR SELECT USING (is_available = true);

-- ══ COMMANDES : chaque user voit seulement ses commandes ══
CREATE POLICY "orders_own" ON orders
  FOR SELECT USING (auth.uid() = user_id);

CREATE POLICY "orders_create" ON orders
  FOR INSERT WITH CHECK (auth.uid() = user_id);

-- ══ LIVREURS : voient leurs propres commandes assignées ══
CREATE POLICY "riders_own_orders" ON orders
  FOR UPDATE USING (
    auth.uid() = (SELECT auth_id FROM riders WHERE id = rider_id)
  );

-- ══ RESTAURANTS : voient leurs propres commandes ══
CREATE POLICY "resto_own_orders" ON orders
  FOR SELECT USING (
    auth.uid() = (SELECT owner_auth_id FROM restaurants WHERE id = restaurant_id)
  );

-- ══ PAIEMENTS : chaque user voit ses paiements ══
CREATE POLICY "payments_own" ON payments
  FOR SELECT USING (auth.uid() = user_id);

-- ══ REALTIME : activer pour le suivi des commandes ══
ALTER PUBLICATION supabase_realtime ADD TABLE orders;
ALTER PUBLICATION supabase_realtime ADD TABLE riders;    -- GPS livreur
